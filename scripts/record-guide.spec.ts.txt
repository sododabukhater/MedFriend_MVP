// Amman = UTC+3. "Today" in the videos is Monday 23 Nov 2026.
const at = (hhmmss: string) => {
  const [h, m, s = 0] = hhmmss.split(':').map(Number)
  return new Date(Date.UTC(2026, 10, 23, h - 3, m, s))
}

const NodeBuffer = (globalThis as unknown as { Buffer: { from(text: string, encoding: string): Uint8Array } }).Buffer

/** Click dots + caption box (both hidden for screenshots). */
async function decorate(page: Page) {
  await page.addInitScript(() => {
    addEventListener(
      'pointerdown',
      (e) => {
        const d = document.createElement('div')
        d.className = 'zz-dot'
        Object.assign(d.style, {
          position: 'fixed', left: `${e.clientX - 22}px`, top: `${e.clientY - 22}px`, width: '44px', height: '44px',
          borderRadius: '50%', background: 'rgba(255,170,0,.45)', border: '3px solid rgba(255,140,0,.9)',
          zIndex: '99998', pointerEvents: 'none', transition: 'opacity .8s, transform .8s',
        })
        document.body.appendChild(d)
        setTimeout(() => { d.style.opacity = '0'; d.style.transform = 'scale(1.6)' }, 250)
        setTimeout(() => d.remove(), 1200)
      },
      true,
    )
  })
}

async function cap(page: Page, text: string, ms = 3200) {
  await page.evaluate((t) => {
    let el = document.getElementById('zz-cap')
    if (!el) {
      el = document.createElement('div')
      el.id = 'zz-cap'
      Object.assign(el.style, {
        position: 'fixed', insetBlockStart: '6px', left: '6px', right: '6px', zIndex: '99999',
        background: 'rgba(16,24,22,.9)', color: '#fff', font: '600 16px/1.35 system-ui, sans-serif',
        padding: '10px 13px', borderRadius: '12px', direction: 'ltr', textAlign: 'left', pointerEvents: 'none',
      })
      document.body.appendChild(el)
    }
    el.style.display = 'block'
    el.textContent = t
  }, text)
  await page.waitForTimeout(ms)
}

async function shot(page: Page, name: string, fullPage = false) {
  await page.evaluate(() => {
    const c = document.getElementById('zz-cap')
    if (c) c.style.display = 'none'
    document.querySelectorAll('.zz-dot').forEach((d) => d.remove())
  })
  await page.screenshot({ path: `${OUT}${name}.png`, fullPage })
  await page.evaluate(() => {
    const c = document.getElementById('zz-cap')
    if (c) c.style.display = 'block'
  })
}

async function finish(page: Page, name: string) {
  await page.waitForTimeout(800)
  const video = page.video()!
  await page.close()
  await video.saveAs(`${OUT}${name}.webm`)
}

/** A pill-like picture drawn in the page. */
async function pillImage(page: Page, kind: 'pill' | 'doc') {
  const dataUrl = await page.evaluate((kind) => {
    const c = document.createElement('canvas')
    c.width = 900
    c.height = 1200
    const x = c.getContext('2d')!
    if (kind === 'pill') {
      x.fillStyle = '#dfe8e4'
      x.fillRect(0, 0, 900, 1200)
      x.fillStyle = '#f4f7f6'
      x.strokeStyle = '#9aa9a3'
      x.lineWidth = 8
      x.beginPath()
      x.ellipse(450, 600, 260, 160, -0.4, 0, Math.PI * 2)
      x.fill()
      x.stroke()
      x.strokeStyle = '#b9c6c1'
      x.beginPath()
      x.moveTo(300, 690)
      x.lineTo(600, 510)
      x.stroke()
      x.fillStyle = '#6b7c75'
      x.font = 'bold 60px sans-serif'
      x.fillText('5', 520, 640)
    } else {
      x.fillStyle = '#ffffff'
      x.fillRect(0, 0, 900, 1200)
      x.fillStyle = '#1a6b5a'
      x.font = 'bold 48px sans-serif'
      x.fillText('City Lab — Blood test', 60, 110)
      x.fillStyle = '#333'
      x.font = '34px sans-serif'
      const rows = [['HbA1c', '6.9 %'], ['Fasting glucose', '128 mg/dL'], ['LDL cholesterol', '96 mg/dL'], ['Creatinine', '0.9 mg/dL']]
      rows.forEach(([a, b], i) => {
        x.fillText(a, 60, 260 + i * 90)
        x.fillText(b, 600, 260 + i * 90)
      })
      x.fillStyle = '#999'
      x.font = '26px sans-serif'
      x.fillText('Sample document for the demo', 60, 1120)
    }
    return c.toDataURL('image/jpeg', 0.9)
  }, kind)
  return { name: kind === 'pill' ? 'pill.jpg' : 'lab.jpg', mimeType: 'image/jpeg', buffer: NodeBuffer.from(dataUrl.split(',')[1], 'base64') }
}

/** A filled account like a real patient: 4 medicines added 30 days ago, a month of history, a name. */
async function patient(page: Page, now: Date, opts: { fakeVoice?: boolean } = {}) {
  await decorate(page)
  if (opts.fakeVoice) {
    // Videos have no sound: the spoken words appear in a yellow speech bubble instead.
    await page.addInitScript(() => {
      const voice = { lang: 'en-US', name: 'English voice', localService: true, default: true, voiceURI: 'English voice' }
      type U = { text: string; onend?: () => void }
      const show = (text: string) => {
        let b = document.getElementById('zz-voice')
        if (!b) {
          b = document.createElement('div')
          b.id = 'zz-voice'
          Object.assign(b.style, {
            position: 'fixed', top: '92px', left: '14px', right: '14px', zIndex: '99999', background: '#fff3b0',
            color: '#3a2e00', border: '2px solid #e0b400', borderRadius: '14px', padding: '10px 12px',
            font: '600 16px/1.35 system-ui, sans-serif', direction: 'ltr', pointerEvents: 'none',
          })
          document.body.appendChild(b)
        }
        b.textContent = `🔊 “${text}”`
        b.style.display = 'block'
      }
      const fake = {
        getVoices: () => [voice],
        speak: (u: U) => {
          show(u.text)
          setTimeout(() => {
            const b = document.getElementById('zz-voice')
            if (b) b.style.display = 'none'
            u.onend?.()
          }, 3500)
        },
        cancel: () => {},
        pause: () => {},
        resume: () => {},
        speaking: false,
        pending: false,
        addEventListener: () => {},
        removeEventListener: () => {},
        onvoiceschanged: null,
      }
      Object.defineProperty(window, 'speechSynthesis', { value: fake, configurable: true })
      class FakeUtterance {
        text: string
        lang = ''
        voice: unknown = null
        rate = 1
        pitch = 1
        volume = 1
        onend: (() => void) | null = null
        onerror: (() => void) | null = null
        onstart: (() => void) | null = null
        constructor(text = '') {
          this.text = text
        }
        addEventListener(type: string, fn: () => void) {
          if (type === 'end') this.onend = fn
        }
        removeEventListener() {}
      }
      Object.defineProperty(window, 'SpeechSynthesisUtterance', { value: FakeUtterance, configurable: true })
    })
  }
  const setAt = new Date(Date.UTC(2026, 10, 23, 3, 0)).toISOString() // 06:00 today
  const user = await loginWithMeds(
    page,
    [
      { name: 'Glucophage', times: ['08:00', '20:00'], strength: [500, 'mg'], createdAt: '2026-10-24T05:00:00Z', stock: { pack: 60, count: 40, setAt } },
      { name: 'Concor', times: ['08:00'], strength: [5, 'mg'], createdAt: '2026-10-24T05:00:00Z', stock: { pack: 30, count: 5, setAt } },
      { name: 'Aspocid', times: ['13:00'], strength: [81, 'mg'], createdAt: '2026-10-24T05:00:00Z' },
      { name: 'Lipitor', times: ['21:00'], strength: [20, 'mg'], createdAt: '2026-10-24T05:00:00Z' },
    ],
    now,
  )
  const { data: meds } = await user.db.from('medications').select('id, name, times')
  let seed = 7
  const rnd = () => ((seed = (seed * 16807) % 2147483647) / 2147483647)
  const logs: object[] = []
  for (let d = 0; d < 30; d++) {
    const day = new Date(Date.UTC(2026, 9, 24 + d)) // 24 Oct .. 22 Nov
    for (const m of meds!) {
      for (const time of m.times as string[]) {
        const r = rnd()
        if (r > 0.93) continue // missed (no answer)
        const [h, min] = time.split(':').map(Number)
        const sched = new Date(Date.UTC(day.getUTCFullYear(), day.getUTCMonth(), day.getUTCDate(), h - 3, min))
        const late = r > 0.85 ? 75 : Math.floor(rnd() * 25)
        logs.push({
          medication_id: m.id,
          scheduled_at: sched.toISOString(),
          status: r > 0.89 && r <= 0.93 ? 'skipped' : 'taken',
          recorded_at: new Date(sched.getTime() + late * 60_000).toISOString(),
        })
      }
    }
  }
  expect((await user.db.from('dose_logs').insert(logs)).error).toBeNull()
  expect((await user.db.auth.updateUser({ data: { full_name: 'Sami Haddad', birth_year: 1962 } })).error).toBeNull()
  await page.reload()
  await expect(page.getByRole('heading', { name: 'Today' })).toBeVisible({ timeout: AUTH_TIMEOUT })
  await page.waitForTimeout(1500)
  return user
}
